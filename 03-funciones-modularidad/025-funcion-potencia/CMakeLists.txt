cmake_minimum_required(VERSION 3.16)
project(funcion_potencia C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Werror -pedantic)
add_library(potencia src/potencia.c)
target_include_directories(potencia PUBLIC include)
add_executable(funcion_potencia src/main.c)
target_link_libraries(funcion_potencia potencia)
enable_testing()
add_executable(test_potencia test/test_potencia.c)
target_link_libraries(test_potencia potencia)
add_test(NAME potencia_contrato COMMAND test_potencia)
target_link_libraries(potencia PUBLIC m)
target_compile_options(test_potencia PRIVATE -UNDEBUG)
