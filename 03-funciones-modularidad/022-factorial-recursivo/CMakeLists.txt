cmake_minimum_required(VERSION 3.16)
project(factorial_recursivo C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Werror -pedantic)
add_library(factorial src/factorial.c)
target_include_directories(factorial PUBLIC include)
add_executable(factorial_recursivo src/main.c)
target_link_libraries(factorial_recursivo factorial)
enable_testing()
add_executable(test_factorial test/test_factorial.c)
target_link_libraries(test_factorial factorial)
add_test(NAME factorial_contrato COMMAND test_factorial)
target_compile_options(test_factorial PRIVATE -UNDEBUG)
