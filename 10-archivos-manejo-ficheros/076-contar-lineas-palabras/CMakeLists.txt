cmake_minimum_required(VERSION 3.16)
project(contar_lineas_palabras C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Werror -pedantic)
add_library(contador src/contar_lineas_palabras.c)
target_include_directories(contador PUBLIC include)
add_executable(contar_lineas_palabras src/main.c)
target_link_libraries(contar_lineas_palabras contador)
enable_testing()
add_executable(test_contador test/test_contador.c)
target_link_libraries(test_contador contador)
add_test(NAME contador_contrato COMMAND test_contador)
target_compile_options(test_contador PRIVATE -UNDEBUG)
