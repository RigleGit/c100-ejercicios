if(BUILD_TESTING)
    add_library(c100_api_debug OBJECT src/compilacion_condicional.c)
    target_include_directories(c100_api_debug PRIVATE include)
    target_compile_definitions(c100_api_debug PRIVATE main=c100_example_main DEBUG _DARWIN_C_SOURCE _DEFAULT_SOURCE)
    target_compile_options(c100_api_debug PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
    set_target_properties(c100_api_debug PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED YES C_EXTENSIONS OFF)
    add_executable(test_contrato_debug tests/test_contrato.c $<TARGET_OBJECTS:c100_api_debug>)
    target_include_directories(test_contrato_debug PRIVATE include)
    target_compile_definitions(test_contrato_debug PRIVATE DEBUG)
    target_compile_options(test_contrato_debug PRIVATE -Wall -Wextra -Werror -pedantic -UNDEBUG)
    target_link_libraries(test_contrato_debug PRIVATE Threads::Threads m)
    add_test(NAME contrato_debug COMMAND test_contrato_debug)
endif()
